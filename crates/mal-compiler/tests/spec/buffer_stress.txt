Tagged :: (Symbol, USize);

word :: (USize, Symbol) -> Symbol := (n, acc) ->
    if (n == 0usize) then acc else word(n - 1usize, acc + "x");

grow :: (Buffer<Tagged>, USize) -> Unit := (items, remaining) ->
    if (remaining == 0usize) then ()
    else {
        length := remaining % 37usize;
        items.new((word(length, ""), length));
        grow(items, remaining - 1usize);
    };

next :: USize -> USize := (state) -> state * 1103515245usize + 12345usize;

churn :: (Buffer<Tagged>, USize, USize, USize) -> Unit := (items, seed, remaining, size) ->
    if (remaining == 0usize) then ()
    else {
        s1 := next(seed);
        s2 := next(s1);
        s3 := next(s2);
        s4 := next(s3);
        operation := (s1 / 65536usize) % 3usize;
        first := (s2 / 65536usize) % size;
        second := (s3 / 65536usize) % size;
        length := (s4 / 65536usize) % 37usize;
        room := size - (if (first > second) then first else second);
        span := if (length > room) then room else length;
        if (operation == 0usize) then items.put(first, (word(length, ""), length))
        else if (operation == 1usize) then items.fill(first, span, (word(length, ""), length))
        else items.copy(first, items, second, span);
        churn(items, s4, remaining - 1usize, size);
    };

broken :: (Buffer<Tagged>, USize, USize) -> USize := (items, index, count) ->
    if (index == #items) then count
    else {
        (label, tag) := items.get(index);
        broken(items, index + 1usize, if (#label == tag) then count else count + 1usize);
    };

main :: Unit -> Int32 := () -> {
    items := make<Tagged>(1usize);
    grow(items, 20000usize);
    churn(items, 7usize, 100000usize, #items);
    if (#items == 20000usize && broken(items, 0usize, 0usize) == 0usize) then 0 else 1
};

    if (index == #items) then 0usize else #(items.get(index)) + total(items, index + 1usize);

### managed_copy_fill_and_put_match_the_canonical_buffer | run:0
word :: (USize, Symbol) -> Symbol := (n, acc) ->
    if (n == 0usize) then acc else word(n - 1usize, acc + "x");

next :: USize -> USize := (state) -> state * 1103515245usize + 12345usize;

init :: (Buffer<Symbol>, Buffer<USize>, USize) -> Unit := (items, model, remaining) ->
    if (remaining == 0usize) then ()
    else {
        length := remaining % 41usize;
        items.new(word(length, ""));
        model.new(length);
        init(items, model, remaining - 1usize);
    };

run :: (Buffer<Symbol>, Buffer<USize>, USize, USize, USize) -> Unit := (items, model, seed, remaining, size) ->
    if (remaining == 0usize) then ()
    else {
        s1 := next(seed);
        s2 := next(s1);
        s3 := next(s2);
        s4 := next(s3);
        operation := (s1 / 65536usize) % 4usize;
        first := (s2 / 65536usize) % size;
        second := (s3 / 65536usize) % size;
        length := (s4 / 65536usize) % 41usize;
        room := size - (if (first > second) then first else second);
        span := if (length > room) then room else length;
        if (operation == 0usize) then { items.put(first, word(length, "")); model.put(first, length); }
        else if (operation == 1usize) then { items.fill(first, span, word(length, "")); model.fill(first, span, length); }
        else { items.copy(first, items, second, span); model.copy(first, model, second, span); };
        run(items, model, s4, remaining - 1usize, size);
    };

mismatches :: (Buffer<Symbol>, Buffer<USize>, USize, USize) -> USize := (items, model, index, count) ->
    if (index == #items) then count
    else mismatches(items, model, index + 1usize, if (#(items.get(index)) == model.get(index)) then count else count + 1usize);

main :: Unit -> Int32 := () -> {
    items := make<Symbol>(0usize);
    model := make<USize>(0usize);
    init(items, model, 5000usize);
    run(items, model, 99usize, 200000usize, 5000usize);
    if (mismatches(items, model, 0usize, 0usize) == 0usize) then 0 else 1
