main :: Unit -> Int32 := () -> [return] => {
    packed := *"mal";
    when (#packed == 0usize) {
        return(1i32);
    };
    value := packed.view<UInt8>(0usize, #packed, (region) -> region.get(0usize));
    if (value == '\x6d') then 0i32 else -1i32;
};
